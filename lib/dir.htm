include "null_file.s7i";
include "osfiles.s7i";


(**
 *  [[file|File]] to read the contents of a directory.
 *)
const type: dirFile is sub null_file struct
    var string: path is "";
    var array string: dirArray is 0 times "";
    var integer: currLine is 0;
    var integer: currColumn is 0;
  end struct;


(**
 *  Open a ''dirFile'' with the given ''dirPath''.
 *  @param dirPath Path of a directory in the OS file system.
 *  @return the ''dirFile'' opened, or [[null_file#STD_NULL|STD_NULL]]
 *          if it could not be opened.
 *  @exception MEMORY_ERROR Not enough memory to convert ''dirPath''
 *             to the system path type, or not enough memory to
 *             store the directory contents.
 *  @exception RANGE_ERROR ''dirPath'' does not use the standard path
 *             representation, or it cannot be converted to the system
 *             path type.
 *)
const func file: openDir (in string: dirPath) is func
  result
    var file: newFile is STD_NULL;
  local
    var dirFile: new_file is dirFile.value;
  begin
    new_file.path := "";
    new_file.currLine := 1;
    new_file.currColumn := 1;
    block
      new_file.dirArray := readDir(dirPath);
      newFile := toInterface(new_file);
    exception
      catch FILE_ERROR:
        newFile := STD_NULL;
    end block;
  end func;


const func file: openDirPath (in string: dirPath) is func
  result
    var file: newFile is STD_NULL;
  local
    var dirFile: new_file is dirFile.value;
  begin
    if dirPath[length(dirPath)] <> '/' then
      new_file.path := dirPath & "/";
    else
      new_file.path := dirPath;
    end if;
    new_file.currLine := 1;
    new_file.currColumn := 1;
    block
      new_file.dirArray := readDir(dirPath);
      newFile := toInterface(new_file);
    exception
      catch FILE_ERROR:
        newFile := STD_NULL;
    end block;
  end func;


(**
 *  Read a file name from a ''dirFile''.
 *  When the function returns inFile.bufferChar contains '\n' or
 *  [[char#EOF|EOF]].
 *  @param inFile Directory file.
 *  @return the file name.
 *)
const func string: getln (inout dirFile: inFile) is func
  result
    var string: stri is "";
  begin
    if inFile.currLine <= length(inFile.dirArray) then
      stri := inFile.path &
          inFile.dirArray[inFile.currLine][inFile.currColumn .. ];
      incr(inFile.currLine);
      inFile.currColumn := 1;
      inFile.bufferChar := '\n';
    else
      inFile.bufferChar := EOF;
    end if;
  end func;


(**
 *  Read a file name from a ''dirFile''.
 *  When the function is left inFile.bufferChar contains '\n' or
 *  [[char#EOF|EOF]].
 *  @param inFile Directory file.
 *  @return the file name.
 *)
const func string: getwd (inout dirFile: inFile) is func
  result
    var string: stri is "";
  begin
    if inFile.currLine <= length(inFile.dirArray) then
      stri := inFile.path &
          inFile.dirArray[inFile.currLine][inFile.currColumn .. ];
      incr(inFile.currLine);
      inFile.currColumn := 1;
      inFile.bufferChar := '\n';
    else
      inFile.bufferChar := EOF;
    end if;
  end func;


const func string: gets (in dirFile: inFile, in integer: maxLength) is func
  result
    var string: stri is "";
  begin
    stri := "";
  end func;


(**
 *  Determine the end-of-file indicator.
 *  @param inFile Directory file.
 *  @return TRUE if the end-of-file indicator is set, FALSE otherwise.
 *)
const func boolean: eof (in dirFile: inFile) is
  return inFile.currLine > length(inFile.dirArray);


(**
 *  Determine if another file name can be read successfully.
 *  This function allows a file to be handled like an iterator.
 *  @param inFile Directory file.
 *  @return FALSE if ''getln'' would return "", TRUE otherwise.
 *)
const func boolean: hasNext (in dirFile: inFile) is
  return inFile.currLine <= length(inFile.dirArray);
